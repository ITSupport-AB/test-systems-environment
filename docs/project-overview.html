Firestore security rules (anti-inflation + ownership)

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --- Helper Functions ---
    function getUserUid() { return request.auth.uid; }
    function getDocUid() { return resource.id; }
    function isOwner() { return getUserUid() == getDocUid(); }
    function isFromAdmin() { return request.auth.token.admin == true; }

    // --- Users: owner-only, VIP fields server-writable only ---
    match /users/{userId} {
      allow read: if isOwner();
      allow create: if isOwner()
        && request.resource.data.keys().hasAll(['createdAt'])
        && !request.resource.data.diff(resource.data).affectedKeys()
          .hasAny(['isVIP', 'vipExpiry', 'vipProductId', 'vipPlatform']);
      allow update: if isOwner()
        && !request.resource.data.diff(resource.data).affectedKeys()
          .hasAny(['isVIP', 'vipExpiry', 'vipProductId', 'vipPlatform',
                   'vipUpdatedAt', 'vipEnvironment', 'vipExpiredAt']);
      allow delete: if false;

      match /data/{docId} { allow read: if isOwner(); allow write: if isOwner(); }
      match /receiptNonces/{nonceId} { allow read: if isOwner(); allow create, update, delete: if false; }
    }

    // --- Leaderboard: monotonic scores, no deletes ---
    match /leaderboard/{seasonId}/entries/{playerId} {
      allow read: if request.auth != null;
      allow create: if request.auth.uid == playerId
        && request.resource.data.keys().hasAll(['playerName', 'score', 'uid', 'submittedAt'])
        && request.resource.data.score is int
        && request.resource.data.score > 0
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.playerName is string
        && request.resource.data.playerName.size() > 0
        && request.resource.data.playerName.size() <= 24;
      allow update: if request.auth.uid == playerId
        && request.resource.data.score >= resource.data.score
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['score', 'playerName', 'submittedAt', 'highWave', 'metadata'])
        && request.resource.data.playerName is string
        && request.resource.data.playerName.size() > 0
        && request.resource.data.playerName.size() <= 24;
      allow delete: if false;
    }

    // --- Global deny-all catch-all ---
    match /{document=**} { allow read, write: if false; }
  }
}
